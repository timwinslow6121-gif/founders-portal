Migration content:
```python
"""humana mbi partial index + unresolvable_json column

Revision ID: 014
Revises: 013
Create Date: 2026-05-07

"""
from alembic import op
import sqlalchemy as sa


revision = '014'
down_revision = '013'
branch_labels = None
depends_on = None


def upgrade():
    # Drop the existing simple unique index on customers.mbi
    op.drop_index('ix_customers_mbi', table_name='customers')

    # Create partial unique index — only enforces uniqueness on non-NULL values
    # NOTE: op.create_index() with postgresql_where MUST be called directly on op,
    # NOT inside batch_alter_table — Alembic constraint.
    op.create_index(
        'ix_customers_mbi',
        'customers',
        ['mbi'],
        unique=True,
        postgresql_where=sa.text('mbi IS NOT NULL'),
    )

    # Add unresolvable_json column to import_batches (used by Plan 04-04)
    op.add_column('import_batches', sa.Column('unresolvable_json', sa.Text(), nullable=True))


def downgrade():
    op.drop_column('import_batches', 'unresolvable_json')

    op.drop_index('ix_customers_mbi', table_name='customers')

    with op.batch_alter_table('customers', schema=None) as batch_op:
        batch_op.create_index('ix_customers_mbi', ['mbi'], unique=True)
```

Per D-02: replace simple unique index with partial unique index `WHERE mbi IS NOT NULL`.
Per RESEARCH.md key constraint #1: `op.create_index()` with `postgresql_where=` must be called directly on `op`, NOT inside `batch_alter_table`.
Per RESEARCH.md Work Area 4: `unresolvable_json TEXT` column is added here so Plan 04-04 doesn't need its own migration.

CRITICAL: do NOT run `flask db upgrade` in this task. The migration must run on VPS only AFTER the backfill script (Task 1) has converted all `mbi=""` to NULL — otherwise the partial index creation will succeed but the next insert with `mbi=""` will violate it.
