```python
@upload_bp.route('/upload/unresolvable/resolve', methods=['POST'])
@login_required
def resolve_unresolvable():
    """Resolve one row from a batch's unresolvable_json list.

    Request body (JSON or form):
      batch_id: int
      row_idx: int — index into the unresolvable_json array
      action: 'assign_existing' | 'enter_mbi' | 'create_new'
      customer_id: int — required for 'assign_existing'
      mbi: str — required for 'enter_mbi' and 'create_new'
    """
    from app.models import ImportBatch, Customer, Policy
    import json

    data = request.get_json(silent=True) or request.form
    batch_id = int(data.get('batch_id', 0))
    row_idx = int(data.get('row_idx', -1))
    action = data.get('action', '')

    batch = ImportBatch.query.filter_by(
        id=batch_id, agency_id=current_user.agency_id
    ).first_or_404()

    if not batch.unresolvable_json:
        return jsonify({'error': 'No unresolvable rows on this batch.'}), 400

    rows = json.loads(batch.unresolvable_json)
    if row_idx < 0 or row_idx >= len(rows):
        return jsonify({'error': 'Invalid row index.'}), 400

    row = rows[row_idx]

    if action == 'assign_existing':
        cid = data.get('customer_id', type=int) if hasattr(data, 'get') else int(data.get('customer_id', 0))
        target = Customer.query.filter_by(
            id=cid, agency_id=current_user.agency_id
        ).first_or_404()
        # Link the policy by setting policy.mbi = target.mbi (if target has an MBI)
        if target.mbi and row.get('member_id'):
            Policy.query.filter_by(
                agency_id=current_user.agency_id,
                carrier=row.get('carrier'),
                member_id=row.get('member_id'),
            ).update({'mbi': target.mbi})

    elif action in ('enter_mbi', 'create_new'):
        mbi = (data.get('mbi') or '').strip().upper()
        if not mbi:
            return jsonify({'error': 'MBI is required.'}), 400

        # Try to match existing customer by MBI first
        existing = Customer.query.filter_by(
            agency_id=current_user.agency_id, mbi=mbi
        ).first()

        if existing and action == 'enter_mbi':
            target_id = existing.id
        else:
            # Create a new customer from the row data
            full_name = row.get('full_name') or ''
            parts = full_name.split(' ', 1)
            first_name = parts[0] if parts else ''
            last_name = parts[1] if len(parts) > 1 else ''
            new_c = Customer(
                agency_id=current_user.agency_id,
                primary_agent_id=current_user.id,
                first_name=first_name,
                last_name=last_name,
                mbi=mbi,
                dob=row.get('dob'),
            )
            db.session.add(new_c)
            db.session.flush()
            target_id = new_c.id

        # Update the corresponding Policy row to carry this MBI
        if row.get('member_id'):
            Policy.query.filter_by(
                agency_id=current_user.agency_id,
                carrier=row.get('carrier'),
                member_id=row.get('member_id'),
            ).update({'mbi': mbi})

    else:
        return jsonify({'error': f'Unknown action: {action}'}), 400

    # Remove the resolved row from unresolvable list
    rows.pop(row_idx)
    batch.unresolvable_json = json.dumps(rows) if rows else None
    db.session.commit()

    return jsonify({'ok': True, 'remaining': len(rows)})
```

Add the `resolveRow(batchId, idx, action)` JS function:
```javascript
async function resolveRow(batchId, idx, action) {
  const row = document.querySelectorAll('.unresolvable-row')[idx];
  const payload = { batch_id: batchId, row_idx: idx, action };
  if (action === 'enter_mbi' || action === 'create_new') {
    const mbiInput = row.querySelector('[data-mbi-input]');
    payload.mbi = mbiInput.value.trim();
    if (!payload.mbi) { alert('Enter MBI first.'); return; }
  } else if (action === 'assign_existing') {
    payload.customer_id = parseInt(row.querySelector('[data-customer-id]').value);
  }
  const res = await fetch('/upload/unresolvable/resolve', {
    method: 'POST',
    headers: {'Content-Type':'application/json'},
    body: JSON.stringify(payload),
  });
  const out = await res.json();
  if (out.ok) {
    row.remove();
    document.getElementById('cnt-unresolvable').textContent = out.remaining || '';
    if (out.remaining === 0) {
      document.getElementById('unresolvable-empty').style.display = '';
    }
  } else {
    alert('Error: ' + (out.error || 'unknown'));
  }
}
```
